\chapter{Notation Glossary}
\label{app:notation}

\section{Matrix Families and Scaling}
\begin{description}
    \item[$d$] Matrix dimension, with $d\geq1$.
    \item[$\mathcal{A}=\{A_1,\dots,A_n\}$] Finite nonempty family of real $d\times d$ matrices.
    \item[$n$] Number of base matrices.
    \item[$\tilde{\mathcal{A}}$] Family scaled by a positive candidate value.
    \item[$\lambda$] Positive averaged spectral radius used in preprocessing.
    \item[$\rho(P)$] Spectral radius of a square matrix $P$.
    \item[$\JSR(\mathcal{A})$] Joint spectral radius of the family.
    \item[$\mathcal{G}=\{G_1,\dots,G_m\}$] Fixed finite generator set, with $\rho(G_i)\leq1$.
    \item[$H_i$] Fixed nonempty positive word defining $G_i=A_{H_i}$.
    \item[$k_i=|H_i|$] Length of the defining generator word.
\end{description}

\section{Words and Matrix Products}
\begin{description}
    \item[$\mathcal{I}=\{1,\dots,n\}$] Positive index alphabet.
    \item[$I\in\mathcal{I}^k$] Positive word of length $k$.
    \item[$A_I=A_{i_k}\cdots A_{i_1}$] Matrix product for $I=[i_1,\dots,i_k]$.
    The first index acts first.
    \item[{$[]$}] Empty word, whose matrix is $I_d$.
    \item[$\mathcal{A}^k$] Set of matrix products from positive words of length $k$.
    \item[$\mathcal{J}=\{-m,\dots,-1,1,\dots,n\}$] Extended index alphabet.
    \item[$J\in\mathcal{J}^l$] Node index of depth $l$, with positive and generator positions.
    \item[$A_{-i}=G_i$] Matrix associated with a negative index.
    \item[$\mathcal{W}(J)$] Positive words obtained by expanding the generator positions of $J$.
    \item[$A_J$] Matrix family represented by an extended node index $J$.
    \item[{$[U,J]$}] Concatenation of words, with nested brackets flattened.
\end{description}

\section{Norms and Bounds}
\begin{description}
    \item[$\mathrm{co}_{\mathrm{s}}(V)$] Symmetrized convex hull $\mathrm{co}(V\cup(-V))$.
    \item[$\lVert\cdot\rVert_V$] Minkowski norm of the balanced polytope generated by a finite spanning set $V$, also used for its induced matrix norm.
    \item[$\lVert L\rVert$] Supremum of the matrix norms over a matrix family $L$.
    \item[$\lVert Lv\rVert$] Supremum of $\lVert Pv\rVert$ over $P\in L$.
    \item[$C_{\mathcal{A}}$] Common constant $\max\{1,\lVert A_1\rVert,\dots,\lVert A_n\rVert\}$.
    \item[$C_{\mathcal{G}}$] Common constant in the generator-power bound $\lVert G_i^r\rVert\leq C_{\mathcal{G}}(1+r)^{d-1}$.
    \item[$h$] Maximum node depth of a fixed finite tree.
    \item[$p_T$] Common polynomial bound for words encoded by nodes of $T$.
    \item[$p$] Common polynomial bound used in the hybrid induction.
\end{description}

\section{Certificates and Algorithms}
\begin{description}
    \item[$V$] Finite spanning set of stored vertices, or the final vertex set after termination.
    \item[$M$] Number of vertices when $V=\{v_1,\dots,v_M\}$.
    \item[$Q$] Queue of pending nodes or vertices, as specified by the algorithm.
    \item[$\gets$] Assignment in pseudocode.
    \item[$\xleftarrow{\cup}$] Union update: for sets $S,U$, the instruction $S\xleftarrow{\cup}U$ means $S\gets S\cup U$.
    \item[$T_{\mathcal{A}}$] Infinite positive-word product tree.
    \item[$\AG$-tree] Finite rooted tree relative to the fixed admissible generators, with internal root and the branching rules of Definition~\ref{def:agTree}.
    \item[Covered node] Node whose expanded word family is contained in that of a proper ancestor.
    \item[$\mathcal{L}(T)$] Matrix families represented by the uncovered leaves of $T$.
    \item[$1$-bounded] Every uncovered leaf family has matrix norm at most $1$.
    \item[$1$-$v$-bounded] Every uncovered leaf family maps the specified vertex $v$ into the norm unit ball.
    \item[PL decomposition] Partition into nonempty uncovered leaf words and a remainder encoded by a node.
    \item[$T_v$] Test tree chosen for vertex $v$.
    \item[$T_{\mathcal{G}}$] Minimal tree for one fixed generator word and the understood base family.
\end{description}
