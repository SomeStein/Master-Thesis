
\chapter{Introduction and Background}
\label{ch:introduction} % for a text reference to this chapter: \ref{ch:introduction}

The \emph{joint spectral radius} (JSR), introduced by \citet{RotaStrang1960NoteJSR}, measures the maximal asymptotic growth of products drawn from a matrix family. It governs stability under arbitrary switching and appears in approximation theory. Exact computation is difficult because the number of candidate products grows exponentially with their length. The invariant-polytope algorithm \citep{GuglielmiProtasov2013ExactComputationJSR} and finite-tree algorithm \citep{MoellerReif2014TreeBasedJSR} address this difficulty with different finite certificates.

This thesis explains the invariant-tree algorithm and its termination certificate using the framework of \citet{MejstrikReif2025HybridApproachJSR}. It makes the common-tree certificate's vertex-wise extension explicit and proves it by induction. Its numerical contribution is a reproducible comparison between ordinary IPA and its strict hybrid extension, including a matrix pair for which the hybrid run closes a test tree while ordinary IPA reaches the prescribed iteration limit. Other theoretical results are attributed reconstructions of the cited literature.

\section{Notation and Preliminaries}
\label{sec:introPreliminaries}

Unless stated otherwise, all matrices are real $d \times d$ matrices and all matrix norms are submultiplicative. We use the following notation conventions consistently across all chapters:
\begin{itemize}
    \item Matrix families and dimensions:
    \[
        \mathcal{A}=\{A_1,\dots,A_n\},\qquad
        \tilde{\mathcal{A}}=\{\tilde{A}_1,\dots,\tilde{A}_n\},\qquad
        d=\text{matrix dimension},\quad n=|\mathcal{A}|.
    \]
    \item Generator notation:
    \[
        \mathcal{G}=\{G_1,\dots,G_m\},\qquad m=|\mathcal{G}|.
    \]
    \item Positive indices and products:
    \[
        \mathcal{I}:=\{1,\dots,n\},\quad
        I=[i_1,\dots,i_k]\in\mathcal{I}^k,\quad
        A_I:=A_{i_k}\cdots A_{i_1},\quad
        \mathcal{A}^k:=\{A_I:I\in\mathcal{I}^k\}.
    \]
    \item Scalars and spectra: $\alpha$ denotes a scalar, $x$ a vector, $\lambda$ an eigenvalue, $\rho(\cdot)$ the spectral radius, and $\sigma(\cdot)$ a singular value.
    \item Polytope norms:
    \[
        \|\cdot\|_{\mathrm{co}_{\mathrm{s}}(V)}=\|\cdot\|_V.
    \]
\end{itemize}

Appendix~\ref{app:notation} contains the full notation glossary used throughout the thesis.

\section{Motivation and Applications of the Joint Spectral Radius}
\label{sec:introMotivationJSR}

Let $\mathcal{A} = \{A_1, \cdots, A_n\}$ be a finite set of $d\times d$ matrices. 
Consider a discrete-time linear system generated by a matrix sequence from $\mathcal{A}$:
\begin{equation} \label{eq:switched_system}
x_{t+1} = B_t x_t, \quad x_0 \in \mathbb{R}^d,
\end{equation}
where $B_t \in \mathcal{A}$ for all $t \in \mathbb{N}$.

A key question is whether the system is \emph{uniformly asymptotically stable}: all trajectories must converge to zero for all initial states and all matrix sequences $(B_t)_{t\ge 0}$ with $B_t \in \mathcal{A}$, with a rate uniform over the sequences:
\[
\lim_{t \to \infty} x_t = 0.
\]
This stability question is answered by a generalization of the spectral radius for a single matrix, the so-called \emph{joint spectral radius (JSR)} of a matrix set $\mathcal{A}$ \citep{RotaStrang1960NoteJSR}.
\begin{definition}\citep{RotaStrang1960NoteJSR}
    \label{def:jsrDefinition}
    For a finite set of matrices $\mathcal{A} = \{A_1, \dots, A_n\}$, the JSR is defined as:
    \begin{equation} \label{eq:jsr_def}
        \JSR(\mathcal{A}) := \lim_{k \to \infty} \max_{P \in \mathcal{A}^k}  \|P\|^{1/k}.
    \end{equation}
    where $\|\cdot\|$ denotes any matrix norm.
\end{definition}
This definition quantifies the maximal asymptotic growth rate of matrix products from $\mathcal{A}$.

\subsection*{Illustrative Example}

Consider the matrix set:
\[
\mathcal{A} = \left\{ A_1 = 
\begin{bmatrix}
0.5 & 1 \\
0 & 0.5
\end{bmatrix}, \quad
A_2 =
\begin{bmatrix}
0.3 & 0 \\
1 & 0.3
\end{bmatrix}
\right\}.
\]
Each matrix individually has spectral radius less than $1$. However, their product
\[
A_1 A_2 =
\begin{bmatrix}
0.5 & 1 \\
0 & 0.5
\end{bmatrix}
\begin{bmatrix}
0.3 & 0 \\
1 & 0.3
\end{bmatrix}
=
\begin{bmatrix}
1.15 & 0.3 \\
0.5 & 0.15
\end{bmatrix}
\]
has spectral radius of approximately
\[
\rho(A_1 A_2) \approx 1.2825 > 1.
\]
This example shows that switching can amplify the effects of individual matrices, leading to divergent behavior even if all single matrices are contractive. Thus, the JSR can exceed the spectral radius of any individual matrix in the set.

The following results explain the role of the JSR in switched systems \citep{Jungers2009JointSpectralRadius}:
\begin{itemize}
    \item If $\JSR(\mathcal{A})<1$, every switching trajectory converges to zero uniformly at an exponential rate.
    \item If $\JSR(\mathcal{A})=1$, boundedness requires a separate argument. Irreducibility is a sufficient condition for uniformly bounded products, but is not necessary.
\end{itemize}
The JSR also appears in subdivision theory \citep{CavarettaDahmenMicchelli1991StationarySubdivision} and wavelet theory \citep{DaubechiesLagarias1992TwoScaleII,Maesumi2000JSRHolderWavelets}. Other applications include coding-theoretic capacity \citep{MoisionOrlitskySiegel2001CodesAvoidDifferences,BlondelJungersProtasov2006CapacityCodes} and discrete mathematics \citep{Protasov2000PartitionFunctionAsymptotic,JungersProtasovBlondel2008IntegerMatrixGrowth,JungersProtasovBlondel2009OverlapFreeWordsSpectra}.

\section{Fundamental Theoretical Results}
\label{sec:introTheoreticalBackground}

\begin{lemma}[Fekete's Lemma]\label{lem:feketeLemma}
Let $(a_n)_{n\ge 1}$ be a sequence of real numbers such that
\begin{equation}\label{eq:subadditive}
a_{m+n} \le a_m + a_n \qquad \text{for all } m,n\ge 1.
\end{equation}
Then the limit
\[
\lim_{n\to\infty}\frac{a_n}{n}
\]
exists in $\R\cup\{-\infty\}$ and equals $\inf_{n\ge 1}a_n/n$.
\end{lemma}

\begin{proof}
Set $a_0:=0$ and $\alpha:=\inf_{n\ge1}a_n/n$. Fix $n\ge1$. For every sufficiently large $k$, write $k=qn+r$ with $q\ge1$ and $0\le r<n$. Subadditivity gives $a_k\le q a_n+a_r$. The finitely many possible remainder terms $a_r$ are bounded, while $q/k$ tends to $1/n$. Hence
\[
  \limsup_{k\to\infty}\frac{a_k}{k}\le\frac{a_n}{n}.
\]
Taking the infimum over $n$ yields $\limsup a_k/k\le\alpha$. By definition, $a_k/k\ge\alpha$ for every $k$, so $\liminf a_k/k\ge\alpha$. The limit is therefore $\alpha$, including the case $\alpha=-\infty$.
\end{proof}

\begin{theorem}\citep{RotaStrang1960NoteJSR,Jungers2009JointSpectralRadius}
    \label{thm:jsrWellDefined}
    The JSR is well-defined and independent of the choice of the matrix norm.
\end{theorem}

\begin{proof}
For a fixed matrix norm, let $M_k:=\max_{P\in\mathcal A^k}\lVert P\rVert$. Submultiplicativity gives $M_{k+\ell}\le M_kM_\ell$. If some $M_k=0$, all sufficiently long products vanish and the limit defining the JSR is zero. Otherwise $\log M_k$ is subadditive, so Lemma~\ref{lem:feketeLemma} shows that $M_k^{1/k}$ has a limit.

For two matrix norms, finite-dimensional norm equivalence gives positive constants $c,C$ such that $c\lVert P\rVert_1\le\lVert P\rVert_2\le C\lVert P\rVert_1$ for every $P\in\R^{d\times d}$. The same inequalities hold for the maxima over $\mathcal A^k$. Taking $k$-th roots and letting $k$ tend to infinity removes the factors $c$ and $C$, so both norms give the same JSR.
\end{proof}

To facilitate the analysis of the algorithms and presentation of the main results, we first establish fundamental properties of the JSR.

\subsection*{Homogeneity of the Joint Spectral Radius}
\begin{proposition}\citep{Jungers2009JointSpectralRadius}
    \label{prop:jsrHomogeneity}
    The JSR is homogeneous, meaning for any set of matrices $\mathcal{A}$ and scalar $\alpha \in \R$ we have
    \begin{equation}
        \JSR(\alpha \mathcal{A}) = |\alpha| \JSR(\mathcal{A}).
    \end{equation}
\end{proposition}
\begin{proof}
    Let $\mathcal{A} = \{A_1, \dots, A_n\}$ and $\alpha \in \mathbb{R}$. Then
    \begin{align*}
        \JSR(\alpha \mathcal{A}) & = \lim_{k \to \infty} \max_{I \in \mathcal{I}^k} \|\alpha A_{i_k} \cdots \alpha A_{i_1}\|^{1/k} \\
        & = |\alpha| \lim_{k \to \infty} \max_{I \in \mathcal{I}^k} \|A_{i_k} \cdots A_{i_1}\|^{1/k} \\
        & = |\alpha| \JSR(\mathcal{A}) \\
        & \qedhere
    \end{align*}

\end{proof}

\subsection*{Berger--Wang Characterization and Three-Member Inequality}
\begin{proposition}\citep{BergerWang1992BoundedSemigroupsMatrices}
    \label{prop:bergerWangCharacterization}
    The $\JSR$ can equivalently be defined as
    $$
    \JSR(\mathcal{A}) = \limsup_{k \to \infty} \max_{P \in \mathcal{A}^k} \rho(P)^{1/k},
    $$
    if $\mathcal{A}$ is a finite (or compact) set.
\end{proposition}

\begin{proposition}\citep{RotaStrang1960NoteJSR,Jungers2009JointSpectralRadius}\label{prop:threeMemberInequality}
Let $\|\cdot\|$ be a matrix norm. Then for every $k\in\mathbb{N}$,
\begin{equation}\label{eq:three_member}
\max_{P\in\mathcal{A}^k}\rho(P)^{1/k}\ \le\ \JSR(\mathcal{A})\ \le\ \max_{P\in\mathcal{A}^k}\|P\|^{1/k}.
\end{equation}
\end{proposition}

\begin{proof}
Fix $k$ and set $M_j:=\max_{Q\in\mathcal A^j}\lVert Q\rVert$. Since $M_{j+\ell}\le M_jM_\ell$, the submultiplicative limit in Theorem~\ref{thm:jsrWellDefined} equals $\inf_{j\ge1}M_j^{1/j}$. In particular, $\JSR(\mathcal A)\le M_k^{1/k}$.

For any $P\in\mathcal A^k$, its $r$-th power lies in $\mathcal A^{kr}$ and $\rho(P)^r=\rho(P^r)\le\lVert P^r\rVert\le M_{kr}$. Taking $(kr)$-th roots and letting $r$ tend to infinity gives $\rho(P)^{1/k}\le\JSR(\mathcal A)$. Maximizing over $P$ proves the other inequality.
\end{proof}

This inequality underlies many computational methods, since it yields computable lower and upper bounds on $\JSR(\mathcal A)$. As $k\to\infty$, the upper bound $\max_{P\in\mathcal A^k}\|P\|^{1/k}$ converges to $\JSR(\mathcal A)$. The lower bounds $\max_{P\in\mathcal A^k}\rho(P)^{1/k}$ need not converge, but their limit superior equals $\JSR(\mathcal A)$ by Proposition~\ref{prop:bergerWangCharacterization}.

\subsection*{Similarity Invariance and Reducibility}

\begin{proposition}\citep{Jungers2009JointSpectralRadius}
    \label{prop:jsrSimilarityInvariance}
    The $\JSR$ is invariant under similarity transformations, i.e., for any invertible matrix $T$ we have
    \begin{equation}
       \JSR(\mathcal{A}) = \JSR(T^{-1}\mathcal{A}T).
    \end{equation}
\end{proposition}

\begin{proof}
    This follows from the definition of the JSR and the properties of the spectral radius.
    \begin{align*}
    \JSR(T^{-1}\mathcal{A}T) & = \limsup_{k \to \infty} \max_{A_i \in \mathcal{A}^k} \rho(T^{-1}A_{i_k}T \cdot ... \cdot T^{-1}A_{i_1}T)^{1/k}\\
    & = \limsup_{k \to \infty} \max_{A_i \in \mathcal{A}^k} \rho(T^{-1}A_{i_k} \cdot ... \cdot A_{i_1}T)^{1/k}\\
    & = \limsup_{k \to \infty} \max_{A_i \in \mathcal{A}^k} \rho(A_{i_k} \cdot ... \cdot A_{i_1})^{1/k}\\
    & = \JSR(\mathcal{A})   
    \end{align*}
\end{proof}

\begin{definition}\citep{Jungers2009JointSpectralRadius}
    \label{def:matrixSetReducibility}
    A set of matrices is called \emph{(commonly) reducible} if there exists a nontrivial subspace of $\mathbb{R}^d$ that is invariant under all matrices in the set. This means there exists a change of basis that block-triangularizes all matrices in $\mathcal{A}$ at the same time. If $\mathcal{A}$ is not reducible, it is called \emph{irreducible}.
\end{definition}

\begin{proposition}\citep{Protasov1997GeneralizedSpectralRadiusGeometric}
    \label{prop:reducibleBlockDecomposition}
    If $\mathcal{A}$ is reducible, then 
    \begin{equation}
        \JSR(\mathcal{A}) = \max \{ \JSR(\mathcal{B}), \JSR(\mathcal{D}) \},
    \end{equation}
    where $\mathcal{B} = \{B_1,\dots, B_n\}$ and $\mathcal{D} = \{D_1,\dots, D_n\}$ are the blocks of the block-triangularized matrices
    \[
        T^{-1}A_jT = 
        \begin{bmatrix}
        B_j & C_j \\
        0   & D_j 
        \end{bmatrix}
    \]
    under some invertible matrix $T$.
\end{proposition}
\begin{example}
    \label{ex:reducibleMatricesBlockForm}
    \[
    A_1 = \begin{bmatrix} 2 & 1 \\ 1 & 2 \end{bmatrix}, \quad
    A_2 = \begin{bmatrix} 3 & 2 \\ 2 & 3 \end{bmatrix}
    \]

    These matrices are reducible because there exists a nontrivial change of basis that transforms them into block-triangular form. The eigenvectors common to both matrices are

    \[
    v_1 = \begin{bmatrix} 1 \\ 1 \end{bmatrix}, \quad
    v_2 = \begin{bmatrix} 1 \\ -1 \end{bmatrix}.
    \]

    Using the change of basis matrix

    \[
    T = \begin{bmatrix} 1 & 1 \\ 1 & -1 \end{bmatrix},
    \]

    we compute the similarity transformations

    \[
    T^{-1} A_1 T = \begin{bmatrix} 3 & 0 \\ 0 & 1 \end{bmatrix}, \quad
    T^{-1} A_2 T = \begin{bmatrix} 5 & 0 \\ 0 & 1 \end{bmatrix}.
    \]

    Since both matrices are upper triangular in this basis, they share a common invariant subspace, proving that they are reducible.
    Now, from Proposition~\ref{prop:reducibleBlockDecomposition}, we know that the $\JSR$ can be calculated as
    \[ 
    \JSR(\mathcal{A}) = \max \{ \JSR(\{3,5\}), \JSR(\{1,1\})\} = 5
    \]
\end{example}

This can be applied iteratively until the sets of blocks are all irreducible.
The problem is thereby split into similar problems of smaller dimension.
For the following considerations we can now assume $\mathcal{A}$ to be irreducible.

\subsection*{Finiteness Property and Spectrum-Maximizing Products}
\begin{definition}\citep{LagariasWang1995FinitenessConjecture}
    \label{def:finitenessProperty}
    A matrix set $\mathcal{A} = \{ A_1, \dots, A_n \}$ is said to possess the \emph{finiteness property} if there exists a finite sequence of matrices $A_{i_1}, \dots, A_{i_k}$ such that
    \begin{equation}
        \JSR(\mathcal{A}) = \rho(A_{i_k} \cdots A_{i_1})^{1/k}.
    \end{equation}
    Any such product is referred to as a \emph{spectrum-maximizing product} (s.m.p.).
\end{definition}
Although the finiteness property does not hold for all matrix sets, it is crucial for the effectiveness and termination of most algorithmic approaches.
This is shown, for example, in \citep{BlondelTheysVladimirov2003CounterexampleFiniteness}.

\subsection*{Computational Complexity of JSR Computation}

The computation of the joint spectral radius (JSR) is theoretically challenging due to several complexity results.

Approximating the JSR is NP-hard in general \citep{TsitsiklisBlondel1997HardToComputeJSR}. Thus one does not expect a polynomial-time algorithm unless $\text{P}=\text{NP}$. For rational matrix families, deciding whether the JSR is at most one is undecidable \citep{BlondelTsitsiklis2000BoundednessUndecidable}. Structured instances can still admit efficient numerical treatment.

\subsection*{S.m.p.-Candidates and Generators}
Approximating the JSR, using the invariant-polytope or the finite-tree algorithms, requires identifying candidate products or \emph{generators} of the matrix set that contribute most significantly to the asymptotic growth rate. These generators are often derived through optimization techniques and their identification is a key step in computational algorithms.

\subsection*{Spectral Gap}
\begin{definition}\citep[Definition 1.9]{MejstrikReif2025HybridApproachJSR}
\label{def:spectralGap}
    A finite matrix family $\mathcal{A}$ is said to possess a \emph{spectral gap} if there exists a constant $\gamma \in [0,1)$ such that for every $k \in \mathbb{N}$ and every product $A_I \in \mathcal{A}^k$,
    \[
    \rho(A_I) = \JSR(\mathcal{A})^k
    \quad \text{or} \quad
    \rho(A_I) \le \gamma\,\JSR(\mathcal{A})^k.
    \]
    Equivalently, every non-s.m.p.\ product is uniformly separated from the maximal growth rate by the factor $\gamma < 1$.
\end{definition}

\begin{remark}
\label{rem:spectralGapHybrid}
    The strict IPA closure characterization requires a spectral gap, uniqueness of the s.m.p.\ candidate up to cyclic permutations and powers, and a simple leading eigenvalue \citep{MejstrikReif2025HybridApproachJSR}.
    Later chapters show how these conditions relate to the algorithms.
    We also note that \citep{GuglielmiProtasov2024SpectralGapFamilyMatrices} studies a related notion of gap as separation between the first and second Lyapunov exponents for matrix families.
\end{remark}

\begin{example}
\label{ex:spectralGap}
    Consider
    \[
    A_1=\begin{bmatrix}2&0\\0&1\end{bmatrix},
    \qquad
    A_2=\begin{bmatrix}1&0\\0&2\end{bmatrix},
    \qquad
    \mathcal{A}=\{A_1,A_2\}.
    \]
    Since $\max\{\rho(A_1),\rho(A_2)\}=2$ and $\|A_1\|_\infty=\|A_2\|_\infty=2$, we get $\JSR(\mathcal{A})=2$.
    Any product $A_I \in \mathcal{A}^k$ has the form
    \[
    A_I = A_1^m A_2^{k-m}
        = \begin{bmatrix}2^m&0\\0&2^{k-m}\end{bmatrix},
    \]
    so
    \[
    \rho(A_I)=2^{\max\{m,k-m\}}.
    \]
    If $m=0$ or $m=k$, then $\rho(A_I)=2^k=\JSR(\mathcal{A})^k$ and $A_I$ is an s.m.p.\ product.
    If $1\le m\le k-1$, then
    \[
    \rho(A_I)\le 2^{k-1}=\tfrac12\,\JSR(\mathcal{A})^k.
    \]
    Hence Definition~\ref{def:spectralGap} holds with $\gamma=\tfrac12$.
\end{example}

\section{Preprocessing and Normalization}
\label{sec:introPreprocessing}
The discussed algorithms use the same basic preprocessing pipeline:

We want to find the JSR of the finite set of matrices $\mathcal{A} = \{A_1, \cdots, A_n\}$.
\begin{enumerate}
    \item \textbf{Assumptions}: $\mathcal{A}$ is irreducible. 
    \item \textbf{Candidates}: Select products $P = A_{i_k} \cdots A_{i_1}$ suspected to maximize the averaged spectral radius, with $\lambda := \rho(P)^{1/k}$.
    \item \textbf{Rescaling}: Transform $\mathcal{A} \to \tilde{\mathcal{A}}$ with $\tilde A_i := \frac{1}{\lambda} A_i$.
    \item \textbf{Verification}: Establish that $\JSR(\tilde{\mathcal{A}}) = 1$ using the three-member inequality. By homogeneity, this is equivalent to $\JSR(\mathcal{A}) = \lambda$.
\end{enumerate}

They differ in the certificate sought at the verification step. IPA constructs a polytope invariant under every scaled matrix. The finite-tree method bounds products through a $1$-bounded tree. The hybrid algorithm applies tree tests to individual vertices of an evolving polytope.
